import com.daniel.expense_tracker.model.Expense;
import com.daniel.expense_tracker.repository.ExpenseRepository;  // IMPORT the interface
import com.daniel.expense_tracker.repository.util.DatabaseConnection;
import java.sql.*;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;
import java.util.Optional;

public class DatabaseExpenseRepository implements ExpenseRepository {  // IMPLEMENT the interface
    
    @Override
    public Expense save(Expense expense) {
        String sql = "INSERT INTO expenses (amount, category, expense_date, description) VALUES (?, ?, ?, ?)";
        
        try (Connection conn = DatabaseConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS)) {
            
            pstmt.setDouble(1, expense.getAmount());
            pstmt.setString(2, expense.getCategory());
            pstmt.setDate(3, Date.valueOf(expense.getDate()));
            pstmt.setString(4, expense.getDescription());
            
            pstmt.executeUpdate();
            
            try (ResultSet rs = pstmt.getGeneratedKeys()) {
                if (rs.next()) {
                    expense.setId(rs.getInt(1));
                }
            }
            
        } catch (SQLException e) {
            e.printStackTrace();
        }
        return expense;
    }

    @Override
    public Optional<Expense> findById(int id) {
        String sql = "SELECT * FROM expenses WHERE id = ?";
        
        try (Connection conn = DatabaseConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql)) {
            
            pstmt.setInt(1, id);
            
            try (ResultSet rs = pstmt.executeQuery()) {
                if (rs.next()) {
                    return Optional.of(new Expense(
                        rs.getInt("id"),
                        rs.getDouble("amount"),
                        rs.getString("category"),
                        rs.getDate("expense_date").toLocalDate(),
                        rs.getString("description")

            }

            e.printStackTrace();
        }
        return Optional.empty();
    }

        } catch (SQLException e) {
            e.printStackTrace();

            pstmt.setString(1, category);
            
            try (ResultSet rs = pstmt.executeQuery()) {

    @Override
    public void saveAll(List<Expense> expenses) {

        try (Connection conn = DatabaseConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS)) {
            
            conn.setAutoCommit(false);  // Start transaction

                pstmt.setString(2, expense.getCategory());
                pstmt.setDate(3, Date.valueOf(expense.getDate()));
                pstmt.setString(4, expense.getDescription());
                pstmt.addBatch();
            }
            
            pstmt.executeBatch();
            conn.commit();  // Commit transaction
            
        } catch (SQLException e) {
            e.printStackTrace();
        }
    }

package com.daniel;

import com.daniel.expense_tracker.model.Expense;
import java.time.LocalDate;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        // COMPOSITION: Program to interface, not implementation
        ExpenseRepository repo = new DatabaseExpenseRepository();  // Using interface type
        
        Scanner sc = new Scanner(System.in);

        System.out.println("--- Daniel's Expense Tracker ---");
        System.out.print("Amount: ");
        double amt = Double.parseDouble(sc.nextLine());
        System.out.print("Category: ");
        String cat = sc.nextLine();
        System.out.print("Desc: ");
        String desc = sc.nextLine();

        // Create and save expense
        Expense expense = new Expense(amt, cat, LocalDate.now(), desc);
        repo.save(expense);  // Using interface method

        System.out.println("\nDATABASE ENTRIES:");
        // Using interface method
        repo.findAll().forEach(e -> System.out.println(e.toTableRow()));

        sc.close();
        repo.close();  // Clean up

package com.daniel.expense_tracker.repository;

public class RepositoryFactory {
    public static ExpenseRepository createRepository(String type) {
        switch(type.toLowerCase()) {
            case "database":
                return new com.daniel.expense_tracker.repository.impl.DatabaseExpenseRepository();
            case "file":
                return new com.daniel.expense_tracker.repository.impl.FileExpenseRepository();
            case "memory":
                return new com.daniel.expense_tracker.repository.impl.InMemoryExpenseRepository();
            default:
                return new com.daniel.expense_tracker.repository.impl.DatabaseExpenseRepository();
        }
    }
}
Then in Main:

java
ExpenseRepository repo = RepositoryFactory.createRepository("database");
This is true composition - programming to interfaces rather than concrete implementations!
